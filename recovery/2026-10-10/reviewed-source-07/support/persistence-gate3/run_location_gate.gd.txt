extends SceneTree
var checks := 0
var failures := 0
func check(value: bool,label: String) -> void:
	checks += 1
	if not value:
		failures += 1
		push_error("FAIL: "+label)
func _initialize() -> void:
	var script = load("res://tests/persistence/test_authority_hostile.gd")
	if script == null or not script.can_instantiate(): check(false,"Store gate compiles")
	else: script.new().run(self)
	print("RESULT checks=%d failures=%d" % [checks,failures])
	quit(0 if failures == 0 else 1)
