func _test_authority_reset() -> void:
	var ui = new_ui()
	var emitted: Array[String] = []
	ui.upgrade_chosen.connect(func(_offer,_option): emitted.append("upgrade"))
	ui.shop_closed.connect(func(): emitted.append("shop_closed"))
	ui.resume_requested.connect(func(): emitted.append("resume"))
	ui.present_progression(progression())
	ui.open_evolution_details("test-evolution")
	check(ui._mode=="evolution_detail","authority reset fixture enters detail")
	ui.reset_obsolete_presentation({},true)
	check(ui._growth_view.is_empty() and ui._detail_id.is_empty() and ui._return_focus_id.is_empty(),"obsolete authority clears growth, detail and return focus")
	check(ui._mode=="title" and ui._checkpoint_available and ui.focused_semantic_id()=="이어서 출격","authoritative explicit resume is reachable")
	check(emitted.is_empty(),"reset emits no gameplay intent or shop-close")
	ui.navigate("cancel")
	check(ui._mode=="title" and ui._growth_view.is_empty(),"cancel cannot reconstruct obsolete mandatory growth")
	ui.accept_released()
	ui.navigate("accept")
	check(emitted==["resume"],"fresh accept emits only resume")
	ui.present_progression(progression())
	ui.open_evolution_details("test-evolution")
	ui.reset_obsolete_presentation(snapshot("dead"),false)
	check(ui._mode=="dead" and ui._detail_id.is_empty() and ui._growth_view.is_empty(),"authority terminal presentation clears obsolete detail")
	await settle()
	check(ui._scroll.follow_focus,"reset restores focus-follow behavior")
	ui.free()

func _test_input_ownership() -> void:
	var ui = new_ui()
	var router = load("res://game/input/input_router.gd").new()
	root.add_child(router); router.set_process(false)
	router.ui_action.connect(ui.navigate)
	await settle()
	check(ui.focused_semantic_id()=="출격","input ownership fixture starts on first button")
	var key:=InputEventKey.new();key.keycode=KEY_DOWN;key.pressed=true
	root.push_input(key)
	router._process(0.016)
	check(ui.focused_semantic_id()=="설정","one native directional event causes exactly one semantic focus step")
	var up:=InputEventKey.new();up.keycode=KEY_DOWN;up.pressed=false
	root.push_input(up);router._process(0.016)
	var intents:Array[String]=[]
	router.ui_action.connect(func(action):intents.append(action))
	var accept:=InputEventJoypadButton.new();accept.button_index=JOY_BUTTON_A;accept.pressed=true;accept.device=4
	router.handle_event(accept);router.handle_event(accept)
	check(intents.count("accept")==1,"held controller accept emits once")
	router.set_touch_vector(Vector2.RIGHT)
	router.set_active(false)
	check(router.sample_move()==Vector2.ZERO and not router._accept_armed,"background ownership clears movement and accept latch")
	router.handle_event(accept)
	check(intents.count("accept")==1,"disabled router ignores accept")
	router.set_active(true)
	router.handle_event(accept)
	check(intents.count("accept")==1,"foreground return does not rearm retained press")
	accept.pressed=false;router.handle_event(accept);accept.pressed=true;router.handle_event(accept)
	check(intents.count("accept")==2,"explicit release rearms controller accept")
	var lost:Array[int]=[]
	router.active_controller_lost.connect(func():lost.append(1))
	router.active_device=4;router.set_touch_vector(Vector2.RIGHT);router._on_connection(4,false)
	check(lost.size()==1 and router.sample_move()==Vector2.ZERO and not router._accept_armed,"active controller disconnect clears retained ownership and emits pause intent")
	router.set_active(false)
	check(not router._enabled,"disabled router relinquishes input ownership")
	var motion:=InputEventJoypadMotion.new()
	motion.device=2;motion.axis=JOY_AXIS_LEFT_X;motion.axis_value=0.1
	router.set_active(true);router.handle_event(motion)
	check(router.sample_move()==Vector2.ZERO,"stick below deadzone produces no movement")
	motion.axis_value=1.0;router.handle_event(motion)
	check(router.sample_move()==Vector2.RIGHT,"full stick retains analog direction")
	router.set_touch_vector(Vector2.DOWN)
	check(router.sample_move()==Vector2.DOWN,"active touch owns movement over retained stick")
	motion.axis_value=1.0;router.handle_event(motion)
	check(router.sample_move()==Vector2.RIGHT and router._touch==Vector2.ZERO,"new active gamepad motion clears touch ownership")
	router.clear_transient_input();intents.clear()
	var dpad:=InputEventJoypadButton.new();dpad.device=2;dpad.button_index=JOY_BUTTON_DPAD_DOWN;dpad.pressed=true
	root.push_input(dpad);router._process(0.01)
	check(intents==["down"],"D-pad viewport input has one directional owner")
	router._process(0.29)
	check(intents==["down"],"held D-pad respects initial repeat delay")
	router._process(0.02)
	check(intents==["down","down"],"held D-pad repeats once after initial delay")
	dpad.pressed=false;root.push_input(dpad);router._process(0.01);router._process(1.0)
	check(intents==["down","down"],"D-pad release stops repeat")
	await settle();router.free();ui.free()
